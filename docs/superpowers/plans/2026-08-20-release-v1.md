# devorder-guide 开源分发（方案 A：标准仓库 + INSTALL.md 自安装协议）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 devorder-guide 从「内部单目录仓库」改造为「Agent Skills 标准公开仓库」，实现普通客户把 GitHub 链接发给 AI 工具（Claude Code / Codex / Cursor / Kimi Code / OpenClaw / WorkBuddy / TRAE / InsCode Desktop）即自动下载安装，且不破坏现有质量门禁。

**Architecture:** 单一内容真源 `skills/devorder-guide/`（Agent Skills 标准 `skills/*/SKILL.md` 布局，被 gh skill / npx skills / 各工具目录扫描统一消费）；根目录 `SKILL.md` 为 CI 生成镜像（服务 OpenClaw / WorkBuddy 的仓库根导入路径）；所有分发产物（plugin 副本、marketplace.json 版本、.skill zip、SHA256SUMS）由 `scripts/make_artifacts.py` 生成并强制一致性，延续仓库既有的「安装版=源码版」分发纪律。`INSTALL.md` 是 AI 自安装协议——客户粘贴链接后 AI 读取它并执行安装。

**Tech Stack:** Python 3.10+（引擎，无第三方依赖）、bash（质量门禁）、PowerShell（Windows 安装器）、GitHub Actions（CI/Release）、YAML frontmatter（SKILL.md，Agent Skills 标准）。

**Spec:** 本文档 §1「设计决策」即规格（方案 A 在 2026-08-20 调研后的落地形式）；背景调研与评分见会话记录（方案 A 90 分）。

## §1 设计决策（规格）

| # | 决策 | 理由 | 否决的替代 |
|---|---|---|---|
| D1 | 单一真源 `skills/devorder-guide/`，所有产物生成而非手编 | gh skill / npx skills 只认 `skills/*/SKILL.md` 布局；延续「分发一致性」红线 | 根目录作真源（破坏 gh/npx 发现机制） |
| D2 | 根 `SKILL.md` = 生成镜像（字节一致，CI 守卫） | OpenClaw `git:owner/repo`、WorkBuddy 从 Git 仓库导入要求仓库根有 SKILL.md | 根放瘦指针（部分工具只读 SKILL.md 单文件，会误装）；符号链接（GitHub 下载不支持） |
| D3 | `src/ configs/ references/ evals/ scripts/(三件套) pyproject.toml` 全部移入包内 | 包自洽：`python -m src.package_skill`、check_all 在包内相对路径成立 | 留根（两套路径维护） |
| D4 | AGENTS.md 双份职责：包内 = 真源自带技能级纪律（随包分发，运行时不加载、无害）；仓库根 = T7 新建仓库级纪律（分发/构建/贡献） | 0.5.26 真源物理自带 AGENTS.md（R52：整体迁入不可剥离）；开源仓库根有协作纪律是惯例，且需声明「真源在 skills/、根文件为生成物」 | 从真源剥离 AGENTS.md（制造「发布版≠用户版」漂移）；根不设 AGENTS.md（开发 AI 缺仓库级指引） |
| D5 | LICENSE 真源在根，make_artifacts.py 复制进包（R53 注：0.5.26 真源自带 LICENSE，与根哈希一致（2026-08-20 核验）——原样迁入不可剥离，build 覆盖复制保证持续一致） | InsCode vendor 署名要求；避免双份手编漂移 | 包内独立 LICENSE |
| D6 | 版本真源 = pyproject.toml；SKILL.md metadata.version + git tag 由 CI 校验相等 | 三处手改必漂移 | 各自手改 |
| D7 | verify_install.sh 默认安装目标保持 `$HOME/.workbuddy/skills/`，新增 `--temp-install` 模式 + `--targets` 多工具循环 | CI 上无真实安装目录，需临时目录模式；多工具验收需要循环 | 单目标 diff |
| D8 | INSTALL.md 是「AI 安装协议」，首屏即给 AI 的执行指令，人类兜底段落放后 | 粘贴链接场景下 AI 先读到协议 | 人类优先排版（AI 易迷失） |

```python
"""

import hashlib
import json

import subprocess
import sys

def version() -> str:
    """版本唯一真源：pyproject.toml（正则解析——tomllib 需 Python 3.11+，违反 ≥3.10 约束）。"""
    import re
    text = (PKG / "pyproject.toml").read_text(encoding="utf-8")
    m = re.search(r'^version\s*=\s*"([^"]+)"', text, re.MULTILINE)
    if not m:
        raise SystemExit("pyproject.toml 中找不到 version")
    return m.group(1)


def check_mirror() -> list[str]:
    errors = []
    truth = (PKG / "SKILL.md").read_bytes()
    mirror = MIRROR.read_bytes()
    if mirror != truth:
        errors.append("根 SKILL.md 与 skills/devorder-guide/SKILL.md 不一致（运行 --build 同步）")
    return errors

    shutil.copyfile(PKG / "SKILL.md", MIRROR)
    print(f"✅ SKILL.md 镜像已同步（{MIRROR}）")

    ap = argparse.ArgumentParser()
    g = ap.add_mutually_exclusive_group(required=True)
    g.add_argument("--check", action="store_true")
    g.add_argument("--build", action="store_true")

        return 1
    print("✅ 镜像一致性通过")
    return 0
```

```python
def _sync_dir(src: Path, dst: Path, excludes: set[str]) -> None:
    """镜像目录（删除 dst 中 src 没有的文件，按相对路径比对）。"""
    src_files = {f.relative_to(src) for f in src.rglob("*") if f.is_file()}
    if dst.exists():
        for f in dst.rglob("*"):
            if f.is_file() and f.relative_to(dst) not in src_files:
                f.unlink()
    for f in src.rglob("*"):
        if f.is_file():
            rel = f.relative_to(src)
            if not any(part in excludes for part in rel.parts):  # 按路径片段排除缓存目录（f.name 对目录名无效，会把 .pyc 复制进副本）
                (dst / rel).parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(f, dst / rel)


def check_artifacts() -> list[str]:
    errors = check_mirror()
    # SKILL.md metadata.version 与 pyproject 必须一致（Release 资产与包内 frontmatter 版本同源，CI 早拦而非等 release tag 才查）
    try:
        meta = (PKG / "SKILL.md").read_text(encoding="utf-8").split("---")[1]
        meta_ver = [l.split(":", 1)[1].strip().strip('"') for l in meta.splitlines()
                    if l.strip().startswith("version")][0]
        if meta_ver != version():
            errors.append(f"SKILL.md metadata.version {meta_ver} ≠ pyproject {version()}")
    except (IndexError, KeyError):
        errors.append("SKILL.md 中找不到 metadata.version")
    if PLUGIN_DIR.exists():
        # 排除缓存目录：与 _sync_dir 的 excludes 一致——否则本地/CI 跑过 `-m src.*` 后
        # PKG/src/__pycache__ 使集合差恒非空，--check 必红（R39：CI 上 check_all 先跑必生成 __pycache__）
        truth_files = {f.relative_to(PKG).as_posix() for f in PKG.rglob("*") if f.is_file()
                       and not any(part in {"__pycache__", ".pytest_cache", ".ruff_cache"} for part in f.relative_to(PKG).parts)}
        plugin_files = {f.relative_to(PLUGIN_DIR).as_posix() for f in PLUGIN_DIR.rglob("*")
                        if f.is_file() and not f.as_posix().startswith(".claude-plugin")}
        if truth_files - plugin_files or plugin_files - truth_files:
            errors.append("plugins/devorder-guide/ 与包内容不一致")
        else:
            pj = PLUGIN_DIR / ".claude-plugin/plugin.json"
            if not pj.exists() or json.loads(pj.read_text(encoding="utf-8")).get("version") != version():
                errors.append("plugin.json 缺失或版本与 pyproject 不一致")
    for mp_path in (MARKETPLACE, ROOT / ".atomcode-plugin/marketplace.json"):
        if not mp_path.exists():
            errors.append(f"缺失 {mp_path.relative_to(ROOT)}（运行 --build）")
            continue
        try:
            mp = json.loads(mp_path.read_text(encoding="utf-8"))
            if mp.get("plugins", [{}])[0].get("version") != version():
                errors.append(f"{mp_path.relative_to(ROOT)} 版本 ≠ {version()}")
        except (json.JSONDecodeError, IndexError, KeyError):
            errors.append(f"{mp_path.relative_to(ROOT)} 无效")
    # --check 只校验「已提交状态」（镜像/双市场索引/插件副本）。dist/ 不入库，CI 全新 checkout 上必然不存在，
    # 产物存在性由 --build 末尾显式断言负责（见 build_artifacts）。
    return errors


def build_artifacts() -> None:
    build_mirror()
    # 1) LICENSE 复制进包（InsCode vendor 署名要求）
    shutil.copyfile(ROOT / "LICENSE", PKG / "LICENSE")
    # 2) plugin 副本 = 包内容 + plugin.json
    if PLUGIN_DIR.exists():
        shutil.rmtree(PLUGIN_DIR)
    _sync_dir(PKG, PLUGIN_DIR, excludes={"__pycache__", ".pytest_cache", ".ruff_cache"})
    (PLUGIN_DIR / ".claude-plugin").mkdir(parents=True, exist_ok=True)
    (PLUGIN_DIR / ".claude-plugin/plugin.json").write_text(
        json.dumps({"name": "devorder-guide", "version": version(),
                    "description": "DevOrder 对话引导 Skill（Agent Skills 标准分发版）",
                    "author": "<owner>", "license": "Proprietary"}, ensure_ascii=False, indent=2),
        encoding="utf-8")
    # 3) marketplace.json（入库文件，构建后需 git 提交）
    MARKETPLACE.parent.mkdir(parents=True, exist_ok=True)
    MARKETPLACE.write_text(json.dumps({
        "name": "devorder-guide-marketplace",
        "owner": {"name": "<owner>", "email": "<email>"},
        "plugins": [{"name": "devorder-guide", "source": "./plugins/devorder-guide", "version": version()}],
    }, ensure_ascii=False, indent=2), encoding="utf-8")
    # 3.5) InsCode Desktop 用 AtomCode 规范目录（与 .claude-plugin 同内容，入库）——InsCode 官方市场仓实测为 .atomcode-plugin/ 前缀
    (ROOT / ".atomcode-plugin").mkdir(parents=True, exist_ok=True)
    shutil.copyfile(MARKETPLACE, ROOT / ".atomcode-plugin/marketplace.json")
    # 4) .skill zip（复用包内打包器）
    subprocess.run([sys.executable, "-m", "src.package_skill", ".", str(DIST)],
                   cwd=PKG, check=True, env={"PYTHONUTF8": "1", **{k: v for k, v in __import__("os").environ.items()}})
    # 5) SHA256SUMS
    (DIST / "SHA256SUMS").write_text(
        f"{hashlib.sha256((DIST / 'devorder-guide.skill').read_bytes()).hexdigest()}  devorder-guide.skill\n",
```

```python
def check_tag_consistency(tag: str | None) -> list[str]:
    """tag(vX.Y.Z) 与 pyproject/SKILL.md 版本必须一致。"""
    errors = []
    if not tag:
        return errors
    ver = version()
    if tag != f"v{ver}":
        errors.append(f"tag {tag} ≠ pyproject version {ver}")
    try:
        meta = (PKG / "SKILL.md").read_text(encoding="utf-8").split("---")[1]
        meta_ver = [l.split(":", 1)[1].strip().strip('"') for l in meta.splitlines()
                    if l.strip().startswith("version")][0]
        if meta_ver != ver:
            errors.append(f"SKILL.md metadata.version {meta_ver} ≠ pyproject {ver}")
    except (IndexError, KeyError):
        errors.append("SKILL.md 中找不到 metadata.version")
    return errors
```
